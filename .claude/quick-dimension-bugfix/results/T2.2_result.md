```csharp
        /// <summary>
        /// [PHASE2_INSTRUMENTATION] Derive provenance label for the projection point.
        /// </summary>
        private static string DeriveProvenance(
            OpeningReferenceInfo referenceInfo,
            OpeningReferencePair fallbackReferences,
            XYZ estimatedLeftPoint,
            XYZ estimatedRightPoint,
            string label)
        {
            // HostWallOpeningGeometry = reference-owned (edge scan resolved the reference + point atomically)
            if (referenceInfo.ReferenceStrategy == QuickDimensionReferenceStrategy.HostWallOpeningGeometry)
                return "Reference-Owned (HostWallOpeningGeometry)";

            // FamilyInstanceLeftRight = proxy-derived (named reference + independent point)
            if (referenceInfo.ReferenceStrategy == QuickDimensionReferenceStrategy.FamilyInstanceLeftRight)
            {
                // Distinguish proxy source by pointer equality
                if (label == "Left")
                {
                    if (referenceInfo.ReferencePoint == fallbackReferences.LeftPoint)
                        return "Proxy-Derived (Fallback Host Wall Left Edge)";
                    if (referenceInfo.ReferencePoint == estimatedLeftPoint)
                        return "Proxy-Derived (Estimated BBox Left Extremum)";
                }
                else if (label == "Right")
                {
                    if (referenceInfo.ReferencePoint == fallbackReferences.RightPoint)
                        return "Proxy-Derived (Fallback Host Wall Right Edge)";
                    if (referenceInfo.ReferencePoint == estimatedRightPoint)
                        return "Proxy-Derived (Estimated BBox Right Extremum)";
                }
                return "Proxy-Derived (Unknown Source)";
            }

            return "Unknown Strategy";
        }
```
